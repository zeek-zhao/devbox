# 门禁文件清单（gate files）
#
# 用途：AI 提交的 PR 如果动了下面任何文件，自动 merge 必须被拒绝。
# 理由：AI 可以同时改「判定标准」和「被测对象」。CI 全绿本身不构成放行证据——
#       它完全可以把门禁改松再顺利通过。这是 Yocto #56 的原样教训。
#
# 用法：wf4.sh merge <owner/name> <pr#> --gate-files scripts/gate-files.txt
# 语法：一行一个 glob（* 可跨目录，bash 的 [[ $f == $pat ]] 语义）；# 开头为注释
# 例外：确需改动时人工评审后加 --force 合并，PR 里必须写明为什么改门禁
#
# 对应需求：devbox-需求文档 R-A-06 / R-A-07

# ① CI 定义：改这里等于改判定标准
.github/workflows/*.yml

# ② 判定实现本身
scripts/run-example.sh
scripts/coverage-check.sh
scripts/build-images.sh
# 版本一致性的判定实现，AI 不许自己改松
scripts/check-version-truth.sh
# 合并前总闸：它决定「什么算绿」。改松了等于绕过所有门禁
scripts/pre-merge.sh
scripts/check-gate-files.sh
# 提交身份判定：改松了就能把 AI 署名或机器邮箱放进历史
scripts/check-authors.sh

# ③ 清单自己（防止「把清单删空」这种绕过）
scripts/gate-files.txt
# 身份白名单：把自己加进去就等于绕过身份门禁
.github/authors-allowlist.txt

# ④ 版本单一真本：改一个版本号会连带改掉所有镜像的内容
images/versions.env
# 项目版本号（打 tag 的判据）：full 档会校验它必须等于 tag 去掉 v 前缀
VERSION
# 预算和它的判定实现都必须挡在 AI 自动合并之外，否则 AI 可以先把预算调大再"通过"
images/budgets.tsv
scripts/budget-check.sh

# ⑤ 镜像安装逻辑：改错了镜像照样能构建成功，只是内容不对（最难发现的一类）
images/scripts/*.sh
# Dockerfile 同理，而且更根本：它决定的是「被测环境是什么」
# （基础镜像、包列表、入口点）。换了基础镜像而用例仍全绿，不等于代码没问题。
images/*/Dockerfile

# ⑥ 换行与可执行位：改错会让 Linux 侧整片脚本失效，且不报错
.gitattributes
.gitignore

# ⑦ 本地钩子与提交身份约束
.githooks/*

# ⑧ 格式化配置：改了它就能让不达标的代码「自动达标」
#    与第④条同类风险 —— 把 ColumnLimit 从 100 改到 200，代码立刻全部合规，
#    fmt-check 依然全绿，但代码其实没被整理过。
.editorconfig
.clang-format
rustfmt.toml
ruff.toml
.prettierrc.json
.prettierignore

# ⑨ 格式化的判定实现本身
scripts/fmt.sh
scripts/check-docs.sh

# ⑩ 语言 -> 镜像层映射：改了它就能让某门语言指向别的镜像，等于换了被测环境
images/lang-images.tsv
